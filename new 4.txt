-- Carts table
CREATE TABLE IF NOT EXISTS carts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NULL,
    session_id VARCHAR(255) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_session (user_id, session_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Cart items table
CREATE TABLE IF NOT EXISTS cart_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cart_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL DEFAULT 1,
    price DECIMAL(10,2) NOT NULL, -- snapshot at add time
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (cart_id) REFERENCES carts(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE,
    UNIQUE KEY unique_cart_product (cart_id, product_id) -- prevent duplicate product entries
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Orders table
CREATE TABLE IF NOT EXISTS orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    order_number VARCHAR(50) UNIQUE NOT NULL,
    user_id INT NULL, -- NULL for guest orders (if we allow guest checkout)
    session_id VARCHAR(255) NULL, -- for guests
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    subtotal DECIMAL(10,2) NOT NULL,
    shipping_total DECIMAL(10,2) DEFAULT 0.00,
    tax_total DECIMAL(10,2) DEFAULT 0.00,
    grand_total DECIMAL(10,2) NOT NULL,
    status ENUM('pending','processing','shipped','delivered','cancelled') DEFAULT 'pending',
    payment_status ENUM('pending','paid','failed') DEFAULT 'pending',
    shipping_address TEXT NOT NULL,
    billing_address TEXT NOT NULL,
    payment_method VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE SET NULL,
    INDEX idx_user (user_id),
    INDEX idx_order_number (order_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Order items table (one order can have items from multiple sellers)
CREATE TABLE IF NOT EXISTS order_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    seller_id INT NOT NULL,
    product_id INT NOT NULL,
    product_name VARCHAR(255) NOT NULL,
    quantity INT NOT NULL,
    price_per_unit DECIMAL(10,2) NOT NULL,
    total_price DECIMAL(10,2) NOT NULL,
    status ENUM('pending','processing','shipped','delivered','cancelled','returned') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (seller_id) REFERENCES sellers(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE,
    INDEX idx_order (order_id),
    INDEX idx_seller (seller_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

// POST /api/seller/products - create a new product
app.post('/api/seller/products', async (req, res) => {
    if (!req.session.userId || req.session.userRole !== 'seller') {
        return res.status(403).json({ error: 'Seller access required' });
    }
    const { name, description, price, quantity, category, image_url } = req.body;
    if (!name || !price) {
        return res.status(400).json({ error: 'Name and price are required' });
    }
    try {
        const [sellerRows] = await pool.promise().query('SELECT id FROM sellers WHERE user_id = ?', [req.session.userId]);
        if (sellerRows.length === 0) {
            return res.status(404).json({ error: 'Seller record not found' });
        }
        const sellerId = sellerRows[0].id;

        const [result] = await pool.promise().query(
            'INSERT INTO products (seller_id, name, description, price, quantity, category, image_url) VALUES (?, ?, ?, ?, ?, ?, ?)',
            [sellerId, name, description || null, price, quantity || 0, category || null, image_url || null]
        );
        res.status(201).json({ message: 'Product created', productId: result.insertId });
    } catch (err) {
        console.error('Error creating product:', err);
        res.status(500).json({ error: 'Database error' });
    }
});



// PUT /api/seller/products/:id - update a product
app.put('/api/seller/products/:id', async (req, res) => {
    if (!req.session.userId || req.session.userRole !== 'seller') {
        return res.status(403).json({ error: 'Seller access required' });
    }
    const productId = req.params.id;
    const { name, description, price, quantity, category, image_url } = req.body;
    try {
        const [sellerRows] = await pool.promise().query('SELECT id FROM sellers WHERE user_id = ?', [req.session.userId]);
        if (sellerRows.length === 0) {
            return res.status(404).json({ error: 'Seller record not found' });
        }
        const sellerId = sellerRows[0].id;

        // Ensure product belongs to this seller
        const [productRows] = await pool.promise().query('SELECT id FROM products WHERE id = ? AND seller_id = ?', [productId, sellerId]);
        if (productRows.length === 0) {
            return res.status(404).json({ error: 'Product not found or access denied' });
        }

        await pool.promise().query(
            'UPDATE products SET name = ?, description = ?, price = ?, quantity = ?, category = ?, image_url = ? WHERE id = ?',
            [name, description, price, quantity, category, image_url, productId]
        );
        res.json({ message: 'Product updated' });
    } catch (err) {
        console.error('Error updating product:', err);
        res.status(500).json({ error: 'Database error' });
    }
});

        // Main function to load dashboard
        async function loadDashboard() {
            try {
                // 1. Fetch user data
                const userRes = await fetch('/api/user');
                if (!userRes.ok) {
                    if (userRes.status === 401) {
                        window.location.href = '/login.html';
                        return;
                    }
                    throw new Error('Failed to fetch user data');
                }
                const user = await userRes.json();

                // 2. Display welcome card
                document.getElementById('welcomeMessage').textContent = `Welcome back, ${user.full_name}!`;
                document.getElementById('userEmail').innerHTML = `<strong>Email:</strong> ${user.email}`;
                document.getElementById('roleBadge').textContent = user.role;
                document.getElementById('welcomeCard').style.display = 'block';

                // 3. Render role-based modules
                const modules = modulesByRole[user.role] || modulesByRole.buyer;
                const container = document.getElementById('modulesContainer');
                container.innerHTML = ''; // Clear any previous content
                modules.forEach(module => {
                    addModuleCard(module.icon, module.title, module.desc, module.link);
                });

                // 4. If user is a seller, add a public store card
                if (user.role === 'seller') {
                    try {
                        const storeRes = await fetch('/api/seller/store');
                        if (storeRes.ok) {
                            const store = await storeRes.json();
                            addModuleCard('🏪', 'My Public Store', 'View your storefront as customers see it.', `/store.html?slug=${store.store_slug}`);
                        }
                    } catch (err) {
                        console.error('Error fetching store info:', err);
                    }
                }

                // 5. If user is a buyer, check seller application status
                if (user.role === 'buyer') {
                    try {
                        const appRes = await fetch('/api/seller/application');
                        if (appRes.ok) {
                            const appData = await appRes.json();
                            if (!appData) {
                                // No application yet
                                addModuleCard('🏪', 'Become a Seller', 'Start selling your products.', '/seller-apply.html');
                            } else if (appData.status === 'pending') {
                                addModuleCard('⏳', 'Application Pending', 'Your seller application is under review.');
                            }
                        }
                    } catch (err) {
                        console.error('Error fetching seller application:', err);
                    }
                }

                container.style.display = 'grid';

                console.error('Dashboard error:', error);
